/* Router. Bookingflyten kjører uten nav og footer — ingenting skal
   konkurrere med steget. Alle andre sider har full navigasjon og footer. */
(function () {
function App() {
  const [route, setRoute] = React.useState("/");
  const go = (href) => { setRoute(href); window.scrollTo({ top: 0 }); };

  React.useEffect(() => {
    if (window.lucide) window.lucide.createIcons({ attrs: { "stroke-width": 1.75 } });
  }, [route]);

  const [path, query] = route.split("?");
  const seg = path.split("/").filter(Boolean);
  const preset = {};
  if (query) for (const kv of query.split("&")) { const [k, v] = kv.split("="); preset[k] = v; }

  if (seg[0] === "booking") {
    return <div className="dg dg-w"><window.DGBooking go={go} preset={preset} /></div>;
  }

  let view;
  if (!seg.length) view = <window.DGForside go={go} />;
  else if (seg[0] === "tjenester") view = <window.DGTjenester go={go} />;
  else if (seg[0] === "tjeneste") view = <window.DGTjenesteDetalj go={go} id={seg[1]} />;
  else if (seg[0] === "avdelinger") view = <window.DGAvdelinger go={go} />;
  else if (seg[0] === "avdeling") view = <window.DGAvdeling go={go} slug={seg[1]} />;
  else if (seg[0] === "om-oss") view = <window.DGOmOss go={go} />;
  else if (seg[0] === "min-side") view = <window.DGMinSide go={go} />;
  else if (seg[0] === "gavekort") view = <window.DGForside go={go} />;
  else view = <window.DGForside go={go} />;

  return (
    <div className="dg dg-w">
      <window.DGNav route={path} go={go} />
      <main><div key={path} className="dg-page">{view}</div></main>
      <window.DGFooter go={go} />
    </div>
  );
}

ReactDOM.createRoot(document.getElementById("app")).render(<App />);
})();
