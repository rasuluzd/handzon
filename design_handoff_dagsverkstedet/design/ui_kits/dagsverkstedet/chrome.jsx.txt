/* On-scroll reveal. Brukes sparsomt — bare på forsidens hovedblokker.
   Enkel rulle-lytter framfor IntersectionObserver: den fanger også hopp i
   rullefeltet, så ingenting kan bli stående usynlig. */
function Reveal({ delay = 0, as: Tag = "div", className = "", children, ...rest }) {
  const ref = React.useRef(null);
  const [shown, setShown] = React.useState(false);
  React.useEffect(() => {
    if (reduced()) { setShown(true); return undefined; }
    const el = ref.current;
    if (!el) return undefined;
    let done = false;
    const check = () => {
      if (done || !ref.current) return;
      if (ref.current.getBoundingClientRect().top < window.innerHeight * 0.9) {
        done = true;
        setShown(true);
        window.removeEventListener("scroll", check);
        window.removeEventListener("resize", check);
      }
    };
    window.addEventListener("scroll", check, { passive: true });
    window.addEventListener("resize", check);
    check();
    return () => {
      done = true;
      window.removeEventListener("scroll", check);
      window.removeEventListener("resize", check);
    };
  }, []);
  return (
    <Tag ref={ref} className={`dg-reveal${shown ? " in" : ""} ${className}`.trim()} style={{ "--d": `${delay}ms` }} {...rest}>
      {children}
    </Tag>
  );
}

/* Teller et beløp opp eller ned når summen endrer seg. */
function useCountUp(target, dur = 340) {
  const [v, setV] = React.useState(target);
  const shown = React.useRef(target);
  React.useEffect(() => { shown.current = v; }, [v]);
  React.useEffect(() => {
    const a = shown.current;
    if (a === target) return undefined;
    if (reduced()) { setV(target); return undefined; }
    const t0 = performance.now();
    let raf;
    const tick = (t) => {
      const p = Math.min(1, (t - t0) / dur);
      setV(Math.round(a + (target - a) * (1 - Math.pow(1 - p, 3))));
      if (p < 1) raf = requestAnimationFrame(tick);
    };
    raf = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(raf);
  }, [target, dur]);
  return v;
}

function Nav({ route, go }) {
  const [open, setOpen] = React.useState(false);
  const [shrunk, setShrunk] = React.useState(false);
  const base = "/" + route.split("/")[1];
  const jump = (href) => (e) => { e.preventDefault(); setOpen(false); go(href); };
  /* Krymper etter 40 px rulling — stilen gjelder bare under 900 px.
     Lytter på både window og document: i innrammede visninger blir
     rulle-hendelsen ikke alltid levert til window. */
  React.useEffect(() => {
    let raf = 0;
    const onScroll = () => {
      cancelAnimationFrame(raf);
      raf = requestAnimationFrame(() => {
        const y = window.scrollY || document.documentElement.scrollTop || 0;
        setShrunk(y > 40);
      });
    };
    onScroll();
    window.addEventListener("scroll", onScroll, { passive: true });
    document.addEventListener("scroll", onScroll, { passive: true });
    return () => {
      cancelAnimationFrame(raf);
      window.removeEventListener("scroll", onScroll);
      document.removeEventListener("scroll", onScroll);
    };
  }, []);
  return (
    <React.Fragment>
      <nav className={`dg-nav${shrunk ? " is-shrunk" : ""}`}>
        <a className="dg-nav__logo" href="/" onClick={jump("/")} aria-label="Handz On Auto Care – til forsiden">
          <img src="../../assets/logo-original.webp" alt="Handz On Auto Care" />
        </a>
        <div className="dg-nav__links">
          {NAV.map((l) => (
            <a key={l.href} href={l.href} onClick={jump(l.href)}
               className={base === l.href || (l.href === "/tjenester" && base === "/tjeneste") || (l.href === "/avdelinger" && base === "/avdeling") ? "on" : ""}
               aria-current={base === l.href ? "page" : undefined}>{l.label}</a>
          ))}
        </div>
        <Btn onClick={() => go("/booking")}>Bestill time</Btn>
        <button type="button" className="dg-nav__burger" aria-label="Åpne meny" aria-expanded={open} onClick={() => setOpen(true)}>
          <span /><span /><span />
        </button>
      </nav>
      {open && (
        <div className="dg-sheet" role="dialog" aria-modal="true" aria-label="Meny">
          <div className="dg-sheet__top">
            <img src="../../assets/logo-original.webp" alt="Handz On Auto Care" />
            <button type="button" className="dg-sheet__x" onClick={() => setOpen(false)} aria-label="Lukk meny">✕</button>
          </div>
          <nav aria-label="Mobilmeny">
            {NAV.concat([{ href: "/gavekort", label: "Gavekort" }]).map((l) => (
              <a key={l.href} href={l.href} onClick={jump(l.href)}>{l.label}</a>
            ))}
          </nav>
          <div className="dg-sheet__foot">
            <Btn size="lg" block onClick={() => { setOpen(false); go("/booking"); }}>Bestill time</Btn>
            <p>14 avdelinger over hele Norge</p>
          </div>
        </div>
      )}
    </React.Fragment>
  );
}
