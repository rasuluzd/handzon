function Blogg({ onBurger, posts, setPosts, toast }) {
  const [q, setQ] = React.useState("");
  const [filter, setFilter] = React.useState("alle");
  const [draft, setDraft] = React.useState(null);
  const [slugLocked, setSlugLocked] = React.useState(true);

  let list = [...posts].sort((a, b) => (a.date < b.date ? 1 : -1));
  if (filter === "publisert") list = list.filter((p) => p.published);
  if (filter === "utkast") list = list.filter((p) => !p.published);
  if (q) {
    const t = q.toLowerCase();
    list = list.filter((p) => (p.title + " " + p.excerpt + " " + p.cat).toLowerCase().includes(t));
  }

  const drafts = posts.filter((p) => !p.published).length;

  const open = (p) => { setDraft({ ...p }); setSlugLocked(true); };
  const close = () => setDraft(null);

  function addNew() {
    setDraft({
      id: "p" + (posts.length + 1), title: "", slug: "", cat: "Bilpleie-guiden", author: "Ove Hagen",
      date: new Date().toISOString().slice(0, 10), published: false, hero: PHOTOS[0], reads: 0,
      excerpt: "", body: "", isNew: true,
    });
    setSlugLocked(false);
  }

  function save(publish) {
    const next = { ...draft, published: publish != null ? publish : draft.published };
    delete next.isNew;
    setPosts((prev) => (prev.some((p) => p.id === next.id) ? prev.map((p) => (p.id === next.id ? next : p)) : [...prev, next]));
    toast(publish
      ? { title: "Innlegget er publisert", text: `«${next.title}» ligger nå på handzon.no/nyheter.` }
      : { variant: "info", title: "Utkast lagret", text: "Innlegget er ikke synlig for kunder ennå." });
    close();
  }

  function remove() {
    setPosts((prev) => prev.filter((p) => p.id !== draft.id));
    toast({ variant: "info", title: "Innlegget er slettet", text: `«${draft.title || "Uten tittel"}» er fjernet.` });
    close();
  }

  function togglePublished(p) {
    setPosts((prev) => prev.map((x) => (x.id === p.id ? { ...x, published: !x.published } : x)));
    toast(p.published
      ? { variant: "info", title: "Innlegget er avpublisert", text: `«${p.title}» vises ikke lenger på nettsiden.` }
      : { title: "Innlegget er publisert", text: `«${p.title}» er nå live.` });
  }

  function wrap(before, after) {
    const ta = document.getElementById("b-body");
    if (!ta) return;
    const { selectionStart: a, selectionEnd: b, value } = ta;
    const next = value.slice(0, a) + before + value.slice(a, b) + (after || "") + value.slice(b);
    setDraft((d) => ({ ...d, body: next }));
    requestAnimationFrame(() => { ta.focus(); ta.setSelectionRange(a + before.length, b + before.length); });
  }

  const words = draft ? draft.body.trim().split(/\s+/).filter(Boolean).length : 0;

  return (
    <React.Fragment>
      <Top title="Blogg og nyheter" sub={`${posts.length} innlegg · ${drafts} ${drafts === 1 ? "utkast" : "utkast"}`} onBurger={onBurger}
           right={<React.Fragment>
             <Seg label="Filter" value={filter} onChange={setFilter}
                  options={[["alle", "Alle"], ["publisert", "Publisert"], ["utkast", "Utkast"]]} />
             <Btn onClick={addNew}><I n="newspaper" s={16} />Nytt innlegg</Btn>
           </React.Fragment>} />

      <div className="ad-body">
        <div className="ad-card" style={{ padding: 14 }}>
          <label className="ad-search">
            <Lupe />
            <input className="ad-input" type="search" value={q} placeholder="Søk i tittel, ingress eller kategori"
                   aria-label="Søk i innlegg" onChange={(e) => setQ(e.target.value)} />
          </label>
        </div>

        {list.length === 0 ? (
          <Empty icon="newspaper" title="Ingen innlegg her" text="Prøv et annet søk, eller skriv et nytt innlegg."
                 action={<Btn onClick={addNew}>Nytt innlegg</Btn>} />
        ) : (
          <div className="ad-card ad-card--flush">
            <div className="ad-scroll">
              <table className="ad-table">
                <thead>
                  <tr><th>Innlegg</th><th>Kategori</th><th>Dato</th><th className="num">Lesninger</th><th>Status</th><th className="act" /></tr>
                </thead>
                <tbody>
                  {list.map((p) => (
                    <tr key={p.id}>
                      <td>
                        <div style={{ display: "flex", alignItems: "center", gap: 12 }}>
                          <img className="ad-thumb" src={D.PHOTOS + p.hero} alt="" />
                          <div style={{ minWidth: 0, maxWidth: "46ch" }}>
                            <div className="nm">{p.title || "Uten tittel"}</div>
                            <div className="meta" style={{ overflow: "hidden", textOverflow: "ellipsis", whiteSpace: "nowrap" }}>/nyheter/{p.slug}</div>
                          </div>
                        </div>
                      </td>
                      <td>{p.cat}</td>
                      <td className="num" style={{ textAlign: "left" }}>{p.date.split("-").reverse().join(".")}</td>
                      <td className="num">{p.reads ? D.nok(p.reads) : "—"}</td>
                      <td>{p.published ? <Tag variant="ok" dot>Publisert</Tag> : <Tag variant="warn">Utkast</Tag>}</td>
                      <td className="act">
                        <div style={{ display: "inline-flex", gap: 8 }}>
                          <Btn variant="g" size="sm" onClick={() => togglePublished(p)}>{p.published ? "Avpubliser" : "Publiser"}</Btn>
                          <Btn variant="s" size="sm" onClick={() => open(p)}>Rediger</Btn>
                        </div>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        )}

        <p className="ad-note">
          Publiserte innlegg havner under «Nyheter» på handzon.no og i «Bilpleie-guiden» hvis kategorien er satt til det. Utkast er bare synlige her.
        </p>
      </div>

      {draft && (
        <Panel
          title={draft.isNew ? "Nytt innlegg" : "Rediger innlegg"}
          sub={draft.isNew ? "Lagres som utkast til du publiserer." : `${draft.cat} · ${draft.published ? "publisert" : "utkast"} · ${words} ord`}
          onClose={close}
          foot={<React.Fragment>
            {!draft.isNew && <Btn variant="d" size="sm" onClick={remove}>Slett</Btn>}
            <Btn variant="s" onClick={() => save(false)} disabled={!draft.title.trim()}>Lagre utkast</Btn>
            <Btn onClick={() => save(true)} disabled={!draft.title.trim() || !draft.excerpt.trim() || !draft.body.trim()}>Publiser</Btn>
          </React.Fragment>}
        >
          <Field label="Tittel" id="b-tittel" help="Sentence case. Konkret framfor kreativ — folk søker på problemet sitt.">
            <input id="b-tittel" className="ad-input" value={draft.title} placeholder="Slik gjør du bilen klar for høsten"
                   onChange={(e) => {
                     const title = e.target.value;
                     setDraft((d) => ({ ...d, title, slug: slugLocked && !d.published ? slugify(title) : d.slug }));
                   }} />
          </Field>

          <Field label="Nettadresse" id="b-slug" help={`handzon.no/nyheter/${draft.slug || "…"}`}>
            <input id="b-slug" className="ad-input" value={draft.slug} placeholder="slik-gjoer-du-bilen-klar-for-hoesten"
                   onChange={(e) => { setSlugLocked(false); setDraft({ ...draft, slug: slugify(e.target.value) }); }} />
          </Field>

          <div className="ad-row2" style={{ marginTop: 16 }}>
            <Field label="Kategori" id="b-kat">
              <select id="b-kat" className="ad-select" style={{ width: "100%" }} value={draft.cat}
                      onChange={(e) => setDraft({ ...draft, cat: e.target.value })}>
                {CATS.map((c) => <option key={c}>{c}</option>)}
              </select>
            </Field>
            <Field label="Publiseringsdato" id="b-dato">
              <input id="b-dato" className="ad-input" type="date" value={draft.date}
                     onChange={(e) => setDraft({ ...draft, date: e.target.value })} />
            </Field>
          </div>

          <Field label="Ingress" id="b-ing" help="Én til to setninger. Dette er teksten som vises i listen og i søk.">
            <textarea id="b-ing" className="ad-textarea" style={{ minHeight: 80 }} value={draft.excerpt}
                      placeholder="Insektrester, kvae og pollen etter sommeren gjør reell skade om de får stå."
                      onChange={(e) => setDraft({ ...draft, excerpt: e.target.value })} />
          </Field>

          <div className="ad-field" style={{ marginTop: 16 }}>
            <label className="ad-label" htmlFor="b-body">Innhold</label>
            <div className="ad-mdbar">
              <button type="button" onClick={() => wrap("## ", "")}>Mellomtittel</button>
              <button type="button" onClick={() => wrap("**", "**")}>Fet</button>
              <button type="button" onClick={() => wrap("*", "*")}>Kursiv</button>
              <button type="button" onClick={() => wrap("\n- ", "")}>Punktliste</button>
              <button type="button" onClick={() => wrap("[", "](https://)")}>Lenke</button>
            </div>
            <textarea id="b-body" className="ad-textarea" value={draft.body}
                      placeholder={"Skriv innlegget her.\n\n## Mellomtittel\nBrødtekst i korte avsnitt."}
                      onChange={(e) => setDraft({ ...draft, body: e.target.value })} />
            <p className="ad-help">{words} ord · ca. {Math.max(1, Math.round(words / 200))} min lesetid. Markdown for mellomtitler, fet og lister.</p>
          </div>

          <div style={{ marginTop: 20, paddingTop: 18, borderTop: "1px solid var(--line)" }}>
            <div className="ad-secttl">Hovedbilde</div>
            <div className="ad-pickers">
              {PHOTOS.map((p) => (
                <button type="button" key={p} className={`ad-picker${draft.hero === p ? " on" : ""}`}
                        onClick={() => setDraft({ ...draft, hero: p })} aria-label={p}>
                  <img src={D.PHOTOS + p} alt="" />
                </button>
              ))}
            </div>
          </div>

          <div style={{ marginTop: 22, paddingTop: 18, borderTop: "1px solid var(--line)" }}>
            <div className="ad-secttl">Slik ser det ut i listen</div>
            <div className="ad-preview">
              <div className="ad-preview__img"><img src={D.PHOTOS + draft.hero} alt="" /></div>
              <div className="ad-preview__bd">
                <Tag>{draft.cat}</Tag>
                <h3>{draft.title || "Uten tittel"}</h3>
                <p>{draft.excerpt || "Ingressen vises her."}</p>
                <p style={{ marginTop: 12, fontSize: 13, color: "var(--body-soft)" }}>
                  {draft.author} · {draft.date.split("-").reverse().join(".")} · {Math.max(1, Math.round(words / 200))} min
                </p>
              </div>
            </div>
          </div>
        </Panel>
      )}
    </React.Fragment>
  );
}

Object.assign(window, { ADBlogg: Blogg, AD_POSTS: POSTS });
})();
